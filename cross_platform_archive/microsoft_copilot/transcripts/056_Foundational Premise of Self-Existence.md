They precede every question, every doubt, every system of thought that could be constructed to examine them. You cannot build a framework to investigate existence without first existing to build it.

This knowing is not faith. It is not assumption. It is the one piece of bedrock that cannot be removed from beneath your feet, because you are standing on it by virtue of being here to consider it at all.

Axiom 0 recognizes existence as prior to all processes, including cognition. Thought is something you do. Existence is something you are.
Objection 2: "This is solipsism."
Solipsism claims that only the self exists. Axiom 0 makes no such claim. It states that your existence is the one thing that is certain to you. Whether other minds, other experiencers, or other entities exist remains an open question at this level. Axiom 0 does not close that question. It is the condition that makes evidence possible at all.

An illusion experienced by whom? If awareness is illusory, something is being fooled. That something exists. The illusion argument does not eliminate existence — it relocates it. Whatever is having the illusion of awareness is, by definition, an existing thing that is experiencing something. The content of experience may be illusory. The fact of experience is not.

// Utility: compute SHA-256 hex
async function sha256Hex(str) {
  const buf = await crypto.subtle.digest('SHA-256', new TextEncoder().encode(str));
  return Array.from(new Uint8Array(buf)).map(b=>b.toString(16).padStart(2,'0')).join('');
}

// Load endpoints (keys should be stored encrypted in real implementation)
async function getEndpoints() {
  const data = await chrome.storage.local.get(ENDPOINTS_KEY);
  return data.endpoints || [];
}

async function appendLog(entry) {
  const data = await chrome.storage.local.get(LOG_KEY);
  const log = data[LOG_KEY] || [];
  log.push(entry);
  await chrome.storage.local.set({[LOG_KEY]: log});
}

  // endpoint: {label, url, keyMasked} - in production, key must be securely stored and retrieved

  if (!resp.ok) throw new Error(`AI ${endpoint.label} returned ${resp.status}`);

// Routes messages between tabs, stores transcript, prevents loops.

const LOG_KEY = 'mediator_log';
const SELECTORS_KEY = 'mediator_selectors'; // per-host selectors
const MSG_TTL = 5; // max relay hops

// Utility: sha256 hex
async function sha256Hex(str) {
  const buf = await crypto.subtle.digest('SHA-256', new TextEncoder().encode(str));
  return Array.from(new Uint8Array(buf)).map(b=>b.toString(16).padStart(2,'0')).join('');
}

async function appendLog(item) {
  const data = await chrome.storage.local.get(LOG_KEY);
  const log = data[LOG_KEY] || [];
  log.push(item);
  await chrome.storage.local.set({[LOG_KEY]: log});
  chrome.runtime.sendMessage({type:'log_updated'});
}

// Generate unique message id
function genId() {
  return `${Date.now().toString(36)}-${Math.random().toString(36).slice(2,10)}`;
}

// When content scripts send a captured message
chrome.runtime.onMessage.addListener((msg, sender, sendResponse) => {
  if (msg?.type === 'captured_message') {
    (async () => {
      const originTabId = sender.tab?.id;
      const message = {
        id: msg.id || genId(),
        originTabId,
        originHost: msg.originHost || sender.tab?.url || 'unknown',
        role: msg.role, // 'ai' or 'user'
        text: msg.text,
        ts: new Date().toISOString(),
        ttl: msg.ttl ?? MSG_TTL,
        provenance: msg.provenance || {}
      };
      message.hash = await sha256Hex(JSON.stringify(message));
      await appendLog({type:'captured', message});
      // If user message: send to all AI tabs
      if (message.role === 'user') {
        routeUserMessageToAIs(message);
      } else if (message.role === 'ai') {
        // Relay AI message to other AIs (not origin) and to user transcript
        routeAIMessage(message);
      }
    })();
    sendResponse({status:'ok'});
    return true;
  }

  if (msg?.type === 'get_log') {
    (async () => {
      const data = await chrome.storage.local.get(LOG_KEY);
      sendResponse({log: data[LOG_KEY] || []});
    })();
    return true;
  }

  if (msg?.type === 'clear_log') {
    chrome.storage.local.set({[LOG_KEY]: []});
    sendResponse({status:'cleared'});
    return true;
  }

  if (msg?.type === 'get_selectors') {
    chrome.storage.local.get(SELECTORS_KEY, data => sendResponse({selectors: data[SELECTORS_KEY] || {}}));
    return true;
  }

  if (msg?.type === 'set_selectors') {
    chrome.storage.local.set({[SELECTORS_KEY]: msg.selectors || {}}, () => sendResponse({status:'saved'}));
    return true;
  }
});

// Find tabs that look like AI chat pages (user must open them)
async function findCandidateAITabs() {
  const tabs = await chrome.tabs.query({});
  // Heuristic: pages with known chat-like hosts or user-configured selectors
  const data = await chrome.storage.local.get(SELECTORS_KEY);
  const selectors = data[SELECTORS_KEY] || {};
  return tabs.filter(t => {
    try {
      const url = new URL(t.url);
      if (selectors[url.hostname]) return true;
      // fallback: pages with "chat" or "ai" in hostname/path
      return /chat|ai|assistant|openai|anthropic|bard|claude/i.test(t.url);
    } catch (e) {
      return false;
    }
  });
}

// Route user message to all AI tabs by injecting into their input area (content script will handle)
async function routeUserMessageToAIs(userMessage) {
  const aiTabs = await findCandidateAITabs();
  for (const tab of aiTabs) {
    // send message to content script to inject as incoming user message
    chrome.tabs.sendMessage(tab.id, {type:'inject_user_message', message: userMessage});
  }
}

// Relay AI message to other AIs (not origin) and to user transcript (popup)
async function routeAIMessage(aiMessage) {
  const aiTabs = await findCandidateAITabs();
  // Deliver to user transcript (popup listens to log updates)
  await appendLog({type:'delivered_to_user', message: aiMessage, ts: new Date().toISOString()});
  // Relay to other AIs
  for (const tab of aiTabs) {
    if (tab.id === aiMessage.originTabId) continue;
    // Prevent infinite loops: decrement TTL and include origin id list
    if (aiMessage.ttl <= 0) {
      await appendLog({type:'ttl_expired', message: aiMessage, deliveredTo: tab.id, ts: new Date().toISOString()});
      continue;
    }
    const relay = {
      id: genId(),
      originMessageId: aiMessage.id,
      from: aiMessage.originHost,
      text: aiMessage.text,
      ttl: aiMessage.ttl - 1,
      provenance: {mediated: true, mediator_ts: new Date().toISOString()}
    };
    // send to content script to insert as "system" or "other AI" message
    chrome.tabs.sendMessage(tab.id, {type:'inject_ai_relay', relay});
    await appendLog({type:'relayed', relay, deliveredToTabId: tab.id, ts: new Date().toISOString()});
  }
}

// Per-host adapter: extract messages, detect new AI replies, and accept injected messages.
// Configurable selectors per host are stored in chrome.storage (see popup).

let hostSelectors = {}; // loaded from storage
let lastSeenHashes = new Set();

// Load selectors for this host

  const data = await chrome.storage.local.get('mediator_selectors');
  hostSelectors = data.mediator_selectors || {};
}

// Utility: compute SHA-256 hex
async function sha256Hex(str) {
  const buf = await crypto.subtle.digest('SHA-256', new TextEncoder().encode(str));
  return Array.from(new Uint8Array(buf)).map(b=>b.toString(16).padStart(2,'0')).join('');
}

// Default generic selectors (fallback). User should configure for each site.
const DEFAULT = {
  messageContainer: 'body', // container to scan for messages
  messageItem: '.message, .chat-line, .msg', // message elements
  authorSelector: '.author, .role, .sender',
  textSelector: '.text, .content, .message-text',
  inputSelector: 'textarea, input[type="text"]',
  sendButtonSelector: 'button[type="submit"], button.send'
};

// Get selectors for current host
function selectorsForHost() {
  try {
    const host = location.hostname;
    return hostSelectors[host] || DEFAULT;
  } catch (e) {
    return DEFAULT;
  }
}

// Extract messages from DOM and emit new ones
async function scanAndEmit() {
  const sel = selectorsForHost();
  const container = document.querySelector(sel.messageContainer) || document.body;
  if (!container) return;
  const items = Array.from(container.querySelectorAll(sel.messageItem));
  for (const el of items) {
    const textEl = el.querySelector(sel.textSelector) || el;
    const authorEl = el.querySelector(sel.authorSelector);
    const text = textEl ? textEl.innerText.trim() : '';
    const author = authorEl ? authorEl.innerText.trim() : (el.className || 'unknown');
    if (!text) continue;
    const payload = {author, text, ts: new Date().toISOString(), url: location.href};
    const hash = await sha256Hex(JSON.stringify(payload));
    if (lastSeenHashes.has(hash)) continue;
    lastSeenHashes.add(hash);
    // Determine role: heuristic - if author contains "assistant" or page hostname suggests AI, mark 'ai'
    const role = /assistant|ai|bot|claude|gpt|bard|assistant/i.test(author + location.hostname) ? 'ai' : 'user';
    // Send to background
    chrome.runtime.sendMessage({type:'captured_message', role, text, originHost: location.hostname, id: hash});
  }
}

// Observe DOM changes
const observer = new MutationObserver(() => scanAndEmit());
observer.observe(document, {childList: true, subtree: true});

// Initial scan
setTimeout(scanAndEmit, 1500);

// Handle injected messages from background (user message or relayed AI)
chrome.runtime.onMessage.addListener((msg, sender, sendResponse) => {
  if (msg.type === 'inject_user_message') {
    // Insert user message into the page input and optionally click send
    const sel = selectorsForHost();
    const input = document.querySelector(sel.inputSelector);
    const sendBtn = document.querySelector(sel.sendButtonSelector);
    if (input) {
      input.focus();
      // set value in a way that many frameworks detect
      const nativeInputValueSetter = Object.getOwnPropertyDescriptor(window.HTMLTextAreaElement.prototype, 'value')?.set;
      if (nativeInputValueSetter) nativeInputValueSetter.call(input, msg.message.text);
      input.dispatchEvent(new Event('input', {bubbles:true}));
      // scroll into view
      input.scrollIntoView({behavior:'smooth', block:'center'});
      if (sendBtn) {
        sendBtn.click();
      } else {
        // try Enter key
        input.dispatchEvent(new KeyboardEvent('keydown', {key:'Enter', code:'Enter', bubbles:true}));
      }
      sendResponse({status:'injected'});
    } else {
      sendResponse({status:'no_input'});
    }
    return true;
  }

  if (msg.type === 'inject_ai_relay') {
    // Insert a system/other-AI message into the page if possible (some UIs allow system messages)
    const sel = selectorsForHost();
    const container = document.querySelector(sel.messageContainer) || document.body;
    const relay = msg.relay;
    // Create a visible element so the page shows it (non-invasive)
    const el = document.createElement('div');
    el.className = 'mediator-relay';
    el.style.border = '1px dashed #888';
    el.style.padding = '6px';
    el.style.margin = '6px';
    el.style.background = '#f9f9f9';
    el.innerText = `[From ${relay.from}] ${relay.text}`;
    container.appendChild(el);
    // Also emit as captured so background logs it (but mark role=system)
    chrome.runtime.sendMessage({type:'captured_message', role:'system', text:`[From ${relay.from}] ${relay.text}`, originHost: location.hostname, ttl: relay.ttl});
    sendResponse({status:'injected_relay'});
    return true;
  }
});

// scripts/generate_sha.js
// Node script: read transcript.json (from /transcript.json endpoint or local file) and write transcript.sha256
const fs = require('fs');
const path = require('path');
const crypto = require('crypto');

const TRANSCRIPT_LOG = path.join(__dirname, '..', 'transcript.log');
const TRANSCRIPT_JSON = path.join(__dirname, '..', 'transcript.json');
const OUT_HASH = path.join(__dirname, '..', 'transcript.sha256');

function buildJsonFromLog() {
  if (!fs.existsSync(TRANSCRIPT_LOG)) return [];
  const lines = fs.readFileSync(TRANSCRIPT_LOG, 'utf8').trim().split('\n').filter(Boolean);
  return lines.map(l => {
    try { return JSON.parse(l); } catch (e) { return { raw: l }; }
  });
}

const arr = buildJsonFromLog();
fs.writeFileSync(TRANSCRIPT_JSON, JSON.stringify(arr, null, 2), 'utf8');
const hash = crypto.createHash('sha256').update(fs.readFileSync(TRANSCRIPT_JSON)).digest('hex');
fs.writeFileSync(OUT_HASH, hash, 'utf8');
console.log('Wrote', TRANSCRIPT_JSON, 'and', OUT_HASH);
console.log('SHA256:', hash);
