package com.weavesight

import android.Manifest
import android.content.pm.PackageManager
import android.graphics.*
import android.os.*
import android.view.*
import android.widget.*
import android.hardware.Camera
import androidx.appcompat.app.AppCompatActivity
import androidx.core.app.ActivityCompat
import androidx.core.content.ContextCompat

class MainActivity : AppCompatActivity() {
    private lateinit var camera: Camera
    private lateinit var preview: SurfaceView
    private lateinit var overlay: View

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_main)

        preview = findViewById(R.id.camera_preview)
        overlay = findViewById(R.id.overlay_view)

        if (ContextCompat.checkSelfPermission(this, Manifest.permission.CAMERA)
            != PackageManager.PERMISSION_GRANTED) {
            ActivityCompat.requestPermissions(this, arrayOf(Manifest.permission.CAMERA), 0)
        } else {
            startCamera()
        }

        overlay.setWillNotDraw(false)
        overlay.invalidate()
    }

    private fun startCamera() {
        camera = Camera.open()
        val holder = preview.holder
        holder.addCallback(object : SurfaceHolder.Callback {
            override fun surfaceCreated(holder: SurfaceHolder) {
                try {
                    camera.setPreviewDisplay(holder)
                    camera.startPreview()
                } catch (e: Exception) {
                    e.printStackTrace()
                }
            }

            override fun surfaceChanged(holder: SurfaceHolder, format: Int, width: Int, height: Int) {}
            override fun surfaceDestroyed(holder: SurfaceHolder) {
                camera.stopPreview()
                camera.release()
            }
        })
    }

    // Example visual overlay draw
    override fun onWindowFocusChanged(hasFocus: Boolean) {
        super.onWindowFocusChanged(hasFocus)
        overlay.invalidate()
    }
}

A `.jar` (Java ARchive) file is essentially:
- A **zip-compressed** file containing **compiled Java/Kotlin bytecode**, along with any necessary metadata (like a manifest), and sometimes resources like images or configs.
- Used to **bundle libraries or applications** into a single file.
- You can think of it like a `.zip` of Java `.class` files and configuration for distribution or execution.

---

### 🤔 What is `gradle-wrapper.jar`?

Specifically:
- `gradle-wrapper.jar` is part of the **Gradle Wrapper**, which lets people **build your app without installing Gradle manually**.
- It **bootstraps** the exact Gradle version listed in `gradle-wrapper.properties`.
- This `.jar` is just the launcher — it fetches and starts the correct Gradle binary from the internet.
